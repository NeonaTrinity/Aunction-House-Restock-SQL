USE acore_characters;

-- ============================================================
-- DML Simulated Player AH Sales - Safe SQL version
--
-- SAFETY REQUIREMENT:
--   Run this script only while ac-worldserver is STOPPED.
--   ac-database should remain running.
--
-- auctionhouse stores an expiration timestamp, not the original
-- 12/24/48-hour posting duration. To avoid treating a brand-new
-- 24h/48h auction as already old, age eligibility is calculated
-- conservatively from the shortest normal WotLK duration (12h).
-- This guarantees an auction is at least min_age_seconds old
-- before the simulated buyer can select it.
-- ============================================================

SET @minimum_auction_duration := 43200;

SET @fake_seller_1 := 2506;
SET @fake_seller_2 := 2507;

SET @budget := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='market_budget_gold') * 10000;
SET @max_sales := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='max_sales_per_run');
SET @min_age := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='min_age_seconds');
SET @sale_chance := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='sale_chance');
SET @max_price_mult := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='max_price_multiplier');
SET @mail_delay := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='mail_delay_seconds');
SET @cut_rate := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='auction_cut_rate');
SET @buyer_hex := (SELECT config_value FROM custom_ah_simulated_sales_config WHERE config_key='simulated_buyer_hex');
SET @unknown_mult := (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='unknown_item_max_multiplier');

DROP TEMPORARY TABLE IF EXISTS ah_sales_candidates;
CREATE TEMPORARY TABLE ah_sales_candidates (
  pick_order INT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  auction_id BIGINT UNSIGNED,
  itemguid BIGINT UNSIGNED,
  itemowner BIGINT UNSIGNED,
  itemEntry INT UNSIGNED,
  item_count INT UNSIGNED,
  buyoutprice BIGINT UNSIGNED,
  deposit BIGINT UNSIGNED,
  auction_age_seconds BIGINT UNSIGNED,
  payout BIGINT UNSIGNED,
  subject_text VARCHAR(255),
  body_text VARCHAR(255)
);

INSERT INTO ah_sales_candidates
(auction_id, itemguid, itemowner, itemEntry, item_count, buyoutprice, deposit, auction_age_seconds, payout, subject_text, body_text)
SELECT
  ah.id,
  ah.itemguid,
  ah.itemowner,
  ii.itemEntry,
  ii.count,
  ah.buyoutprice,
  ah.deposit,
  GREATEST(0, UNIX_TIMESTAMP() - (ah.time - @minimum_auction_duration)),
  GREATEST(0, ROUND(ah.buyoutprice - (ah.buyoutprice * @cut_rate) + ah.deposit)),
  CONCAT(ii.itemEntry, ':0:2:', ah.id, ':', ii.count),
  CONCAT(@buyer_hex, ':', ah.buyoutprice, ':', ah.buyoutprice, ':', ah.deposit, ':', ROUND(ah.buyoutprice * @cut_rate), ':0:0')
FROM auctionhouse ah
JOIN item_instance ii ON ii.guid = ah.itemguid
JOIN characters seller_char ON seller_char.guid = ah.itemowner
JOIN acore_world.item_template it ON it.entry = ii.itemEntry
LEFT JOIN character_inventory ci ON ci.item = ah.itemguid
LEFT JOIN mail_items mi ON mi.item_guid = ah.itemguid
LEFT JOIN custom_ah_item_pool p ON p.item_entry = ii.itemEntry
LEFT JOIN custom_ah_price_overrides o ON o.item_entry = ii.itemEntry
LEFT JOIN custom_ah_excluded_items x ON x.item_entry = ii.itemEntry
WHERE ah.itemowner NOT IN (@fake_seller_1, @fake_seller_2)
  AND ii.owner_guid = ah.itemowner
  AND ci.item IS NULL
  AND mi.item_guid IS NULL
  AND ah.time > UNIX_TIMESTAMP()
  AND ah.buyoutprice > 0
  AND COALESCE(p.enabled, 1) = 1
  AND x.item_entry IS NULL
  AND GREATEST(0, UNIX_TIMESTAMP() - (ah.time - @minimum_auction_duration)) >= @min_age
  AND RAND() < @sale_chance
  AND ah.buyoutprice <= COALESCE(
    ROUND(o.buyout_override * @max_price_mult),
    ROUND(it.SellPrice * ii.count * COALESCE(p.max_multiplier, @unknown_mult) * @max_price_mult)
  )
ORDER BY GREATEST(0, UNIX_TIMESTAMP() - (ah.time - @minimum_auction_duration)) DESC, RAND();

DROP TEMPORARY TABLE IF EXISTS ah_sales_to_process;

-- Deterministic MySQL 8 greedy selection.
-- This preserves the old behavior: candidates are considered in pick_order;
-- an auction that does not fit the remaining budget is skipped, and a cheaper
-- later auction may still be selected. No same-statement user-variable
-- read/write ordering is used.
SET @old_cte_max_recursion_depth := @@SESSION.cte_max_recursion_depth;
SET SESSION cte_max_recursion_depth = 100000;

CREATE TEMPORARY TABLE ah_sales_to_process
WITH RECURSIVE budget_walk AS (
  SELECT
    c.pick_order,
    c.auction_id,
    c.itemguid,
    c.itemowner,
    c.itemEntry,
    c.item_count,
    c.buyoutprice,
    c.deposit,
    c.auction_age_seconds,
    c.payout,
    c.subject_text,
    c.body_text,
    CAST(
      CASE
        WHEN @max_sales > 0 AND c.buyoutprice <= @budget THEN c.buyoutprice
        ELSE 0
      END AS DECIMAL(20,0)
    ) AS spent,
    CAST(
      CASE
        WHEN @max_sales > 0 AND c.buyoutprice <= @budget THEN 1
        ELSE 0
      END AS UNSIGNED
    ) AS sold_count,
    CAST(
      CASE
        WHEN @max_sales > 0 AND c.buyoutprice <= @budget THEN 1
        ELSE 0
      END AS UNSIGNED
    ) AS take_this
  FROM ah_sales_candidates c
  WHERE c.pick_order = 1

  UNION ALL

  SELECT
    c.pick_order,
    c.auction_id,
    c.itemguid,
    c.itemowner,
    c.itemEntry,
    c.item_count,
    c.buyoutprice,
    c.deposit,
    c.auction_age_seconds,
    c.payout,
    c.subject_text,
    c.body_text,
    w.spent +
      CASE
        WHEN w.sold_count < @max_sales
         AND c.buyoutprice <= (@budget - w.spent)
        THEN c.buyoutprice
        ELSE 0
      END AS spent,
    w.sold_count +
      CASE
        WHEN w.sold_count < @max_sales
         AND c.buyoutprice <= (@budget - w.spent)
        THEN 1
        ELSE 0
      END AS sold_count,
    CASE
      WHEN w.sold_count < @max_sales
       AND c.buyoutprice <= (@budget - w.spent)
      THEN 1
      ELSE 0
    END AS take_this
  FROM budget_walk w
  JOIN ah_sales_candidates c
    ON c.pick_order = w.pick_order + 1
)
SELECT
  pick_order,
  auction_id,
  itemguid,
  itemowner,
  itemEntry,
  item_count,
  buyoutprice,
  deposit,
  auction_age_seconds,
  payout,
  subject_text,
  body_text,
  (@budget - spent) AS remaining_budget,
  sold_count AS sale_number
FROM budget_walk
WHERE take_this = 1
ORDER BY pick_order;

SET SESSION cte_max_recursion_depth = @old_cte_max_recursion_depth;

SET @base_mail_id := GREATEST(
  (SELECT config_value + 0 FROM custom_ah_simulated_sales_config WHERE config_key='custom_mail_id_start'),
  (SELECT IFNULL(MAX(id), 0) + 1000 FROM mail)
);

SELECT
  ah.id AS auction_id,
  COALESCE(ch.name, CONCAT('GUID ', ah.itemowner)) AS seller,
  ii.itemEntry,
  it.name AS item_name,
  ii.count AS stack_size,
  ah.buyoutprice,
  COALESCE(
    ROUND(o.buyout_override * @max_price_mult),
    ROUND(it.SellPrice * ii.count * COALESCE(p.max_multiplier, @unknown_mult) * @max_price_mult)
  ) AS fair_max_price,
  ROUND((GREATEST(0, UNIX_TIMESTAMP() - (ah.time - @minimum_auction_duration))) / 60) AS age_minutes,
  CASE
    WHEN ah.itemowner IN (@fake_seller_1, @fake_seller_2) THEN 'FAKE SELLER - IGNORE'
    WHEN ii.owner_guid <> ah.itemowner THEN 'UNSAFE: ITEM OWNER MISMATCH'
    WHEN ci.item IS NOT NULL THEN 'UNSAFE: ITEM ALSO IN INVENTORY'
    WHEN mi.item_guid IS NOT NULL THEN 'UNSAFE: ITEM ALSO IN MAIL'
    WHEN ch.guid IS NULL THEN 'UNSAFE: SELLER CHARACTER MISSING'
    WHEN ah.time <= UNIX_TIMESTAMP() THEN 'EXPIRED - LEAVE FOR CORE'
    WHEN ah.buyoutprice <= 0 THEN 'NO BUYOUT'
    WHEN x.item_entry IS NOT NULL THEN 'EXCLUDED ITEM'
    WHEN COALESCE(p.enabled, 1) = 0 THEN 'DISABLED ITEM'
    WHEN GREATEST(0, UNIX_TIMESTAMP() - (ah.time - @minimum_auction_duration)) < @min_age THEN 'TOO NEW'
    WHEN ah.buyoutprice > COALESCE(
      ROUND(o.buyout_override * @max_price_mult),
      ROUND(it.SellPrice * ii.count * COALESCE(p.max_multiplier, @unknown_mult) * @max_price_mult)
    ) THEN 'PRICE TOO HIGH'
    WHEN ah.id IN (SELECT auction_id FROM ah_sales_to_process) THEN 'SOLD THIS RUN'
    WHEN p.item_entry IS NULL THEN 'ELIGIBLE UNKNOWN ITEM BUT NOT PICKED'
    ELSE 'ELIGIBLE BUT NOT PICKED'
  END AS result
FROM auctionhouse ah
JOIN item_instance ii ON ii.guid = ah.itemguid
JOIN acore_world.item_template it ON it.entry = ii.itemEntry
LEFT JOIN character_inventory ci ON ci.item = ah.itemguid
LEFT JOIN mail_items mi ON mi.item_guid = ah.itemguid
LEFT JOIN custom_ah_item_pool p ON p.item_entry = ii.itemEntry
LEFT JOIN custom_ah_price_overrides o ON o.item_entry = ii.itemEntry
LEFT JOIN custom_ah_excluded_items x ON x.item_entry = ii.itemEntry
LEFT JOIN characters ch ON ch.guid = ah.itemowner
WHERE ah.itemowner NOT IN (@fake_seller_1, @fake_seller_2)
ORDER BY result, age_minutes DESC, ah.id;

START TRANSACTION;

INSERT INTO mail
(id, messageType, stationery, mailTemplateId, sender, receiver, subject, body, has_items, expire_time, deliver_time, money, cod, checked)
SELECT
  @base_mail_id + sale_number,
  2,
  62,
  0,
  7,
  itemowner,
  subject_text,
  body_text,
  0,
  UNIX_TIMESTAMP() + 2592000,
  UNIX_TIMESTAMP() + @mail_delay,
  payout,
  0,
  4
FROM ah_sales_to_process;

DELETE ah
FROM auctionhouse ah
JOIN ah_sales_to_process s ON s.auction_id = ah.id;

DELETE ii
FROM item_instance ii
JOIN ah_sales_to_process s ON s.itemguid = ii.guid;

COMMIT;

SELECT
  s.auction_id,
  ch.name AS seller,
  s.itemEntry,
  it.name AS item_name,
  s.item_count,
  s.buyoutprice,
  s.payout,
  s.remaining_budget
FROM ah_sales_to_process s
LEFT JOIN characters ch ON ch.guid = s.itemowner
JOIN acore_world.item_template it ON it.entry = s.itemEntry
ORDER BY s.sale_number;

SELECT
  COUNT(*) AS simulated_sales,
  SUM(buyoutprice) AS market_spent_copper,
  SUM(payout) AS player_payout_copper
FROM ah_sales_to_process;
